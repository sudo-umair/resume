% Layout for a one-page, ATS-friendly résumé.
% Single column, real text only (no icons, tables or text in graphics),
% so applicant tracking systems can parse every line.

\documentclass[10pt]{article}

% --- Page ---------------------------------------------------------------
\usepackage[a4paper, margin=13mm, top=11mm, bottom=11mm]{geometry}
\pagestyle{empty}
\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt}
\raggedright

% --- Type ---------------------------------------------------------------
\usepackage{fontspec}
% Fira Sans: tested to extract plain ASCII hyphens and ligatures as letters,
% so "Mid-Senior" or "React-Native" match exactly in ATS keyword searches.
\usepackage[sfdefault, lining]{FiraSans}  % lining figures: even-height dates
\usepackage{xcolor}
\definecolor{accent}{HTML}{1F2FB8}   % deep blue, prints well in greyscale
\definecolor{muted}{HTML}{4A4F5C}

% --- Links and PDF metadata (ATS and search read these) ----------------
\usepackage[hidelinks]{hyperref}
\hypersetup{
  pdftitle    = {Muhammad Umair - Résumé},
  pdfauthor   = {Muhammad Umair},
  pdfsubject  = {Full Stack Engineer - React, Next.js, React Native, Node},
  pdfkeywords = {React, Next.js, React Native, TypeScript, Node.js, Full Stack},
}

% --- Section headings ---------------------------------------------------
\usepackage{titlesec}
% Real uppercase rather than small caps: small-caps glyphs extract as
% mixed case ("EXPERiENCE"), which can break ATS section detection.
\titleformat{\section}
  {\color{accent}\bfseries\addfontfeature{LetterSpace=4}}
  {}{0pt}{\MakeUppercase}
  [\vspace{-0.65em}{\color{accent}\rule{\linewidth}{0.6pt}}]
\titlespacing*{\section}{0pt}{7pt}{4pt}

% --- Lists --------------------------------------------------------------
\usepackage{enumitem}
\setlist[itemize]{leftmargin=1.1em, labelsep=0.45em, itemsep=0.6pt, parsep=0pt, topsep=1.5pt, partopsep=0pt}
\renewcommand{\labelitemi}{\textbullet}

% --- Entries ------------------------------------------------------------
% \role{Title}{Organisation}{Location}{Dates}
\newcommand{\role}[4]{%
  \par\vspace{3pt}%
  {\bfseries #1}\hspace{0.4em}{\color{muted}\textbar}\hspace{0.4em}#2\hfill{\color{muted}#3\hspace{0.6em}\textbar\hspace{0.6em}#4}\par
}

% \project{Name}{Role}{Stack}{Link or empty}
\newcommand{\project}[4]{%
  \par\vspace{3pt}%
  {\bfseries #1}\hspace{0.4em}{\color{muted}\textbar}\hspace{0.4em}{\itshape #2}\hfill{\color{muted}#4}\par
  {\small\color{muted}#3}\par
}

% \skill{Area}{List}
\newcommand{\skill}[2]{{\bfseries #1:}\hspace{0.35em}#2\par}
